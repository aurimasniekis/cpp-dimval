function(dimval_add_example target src)
    add_executable(${target} ${src})
    target_link_libraries(${target} PRIVATE dimval::dimval)
    dimval_set_warnings(${target} PRIVATE)
    dimval_enable_sanitizers(${target})
endfunction()

dimval_add_example(dimval_basic_unit        basic_unit.cpp)

dimval_add_example(dimval_list_catalog      list_catalog.cpp)

dimval_add_example(dimval_mechanics         mechanics.cpp)
dimval_add_example(dimval_error_handling    error_handling.cpp)
